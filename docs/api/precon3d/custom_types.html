Type definitions and data structures for the precon3d image processing and stitching pipeline. This module defines NamedTuples for configuration settings, image metadata, tile and mosaic data, and related utilities.

309        detector_binning: Binning setting.

321    illumination_intensity: str | None = None
322    detector_binning: str | None = None
323
324    def __str__(self) -> str:
325        """Formatted channel metadata."""
326        return (

366class CZIMetadata(NamedTuple):
367    """
368    Comprehensive metadata extracted from a CZI file.
369
370    Attributes:
371        filename: Base CZI filename.
372        author: File author.
373        date_created: ISO timestamp of creation.
374        dims: Dimension order string (e.g., 'XYZCT').
375        dims_shape: Tuple of dimension sizes.
376        scene_names: List of scene identifiers.
377        scene_metadata: List of per-scene metadata.
378        scene_shapes_are_consistent: True if all scenes share the same shape.
379        channel_names: List of channel identifiers.
380        channel_metadata: List of per-channel metadata.
381        tile_width: Width of each tile in pixels.
382        tile_height: Height of each tile in pixels.
383        tile_overlap: Overlap between tiles as string or number.
384        tile_anchor_mode: Anchor mode description.
385        tile_scan_mode: Scan mode description.
386        microscope_name: Microscope make/model.
387        microscope_type: Microscope type.
388        camera_name: Camera model.
389        camera_adapter: Adapter description.
390        camera_pixel_size: Pixel size string.
391        camera_pixel_unit: Pixel size unit.
392        objective_name: Objective lens name.
393        objective_magnification: Magnification string.
394        effective_pixel_size: Effective pixel size.
395        color_mode: Color mode description.
396    """
397
398    filename: str | None = None
399    author: str | None = None
400    date_created: str | None = None
401    dims: str | None = None
402    dims_shape: str | None = None
403    scene_names: list[str] | None = None
404    scene_metadata: list[SceneMetadata] | None = None
405    scene_shapes_are_consistent: bool | None = None
406    channel_names: list[str] | None = None
407    channel_metadata: list[ChannelMetadata] | None = None
408    tile_width: Union[int, str] | None = None
409    tile_height: Union[int, str] | None = None
410    tile_overlap: Union[int, str] | None = None
411    tile_anchor_mode: str | None = None
412    tile_scan_mode: str | None = None
413    microscope_name: str | None = None
414    microscope_type: str | None = None
415    camera_name: str | None = None
416    camera_adapter: str | None = None
417    camera_pixel_size: Union[int, str] | None = None
418    camera_pixel_unit: str | None = None
419    objective_name: str | None = None
420    objective_magnification: Union[int, str] | None = None
421    effective_pixel_size: Union[int, str] | None = None
422    color_mode: str | None = None
423
424    def __str__(self) -> str:
425        """
426        Return a human-readable summary of the CZI metadata.
427
428        Converts the ISO timestamp to local mountain time.
429        """
430        # Parse and convert creation time
431        created = ""
432        if self.date_created:
433            try:
434                dt_utc = datetime.strptime(
435                    self.date_created[:19] + "Z", "%Y-%m-%dT%H:%M:%S%z"
436                )
437                local = dt_utc.astimezone(pytz.timezone("America/Denver"))
438                created = local.strftime("%Y-%m-%d %H:%M:%S %Z")
439            except Exception:
440                created = self.date_created
441
442        # Build scene and channel blocks
443        scenes = "\n".join(str(md) for md in (self.scene_metadata or []))
444        channels = "\n".join(str(md) for md in (self.channel_metadata or []))
445
446        return (
447            f"{self.filename}.czi by {self.author} on {created}\n"
448            f"Dimensions: {self.dims} shape={self.dims_shape}, "
449            f"consistent_shapes={self.scene_shapes_are_consistent}\n"
450            f"Tiles: {self.tile_width}×{self.tile_height}, overlap={self.tile_overlap}, "
451            f"anchor={self.tile_anchor_mode}, scan_mode={self.tile_scan_mode}\n"
452            f"Scenes ({len(self.scene_names or [])}): {self.scene_names}\n"
453            f"Channels ({len(self.channel_names or [])}): {self.channel_names}\n"
454            f"\nScene Details:\n{scenes}\n\nChannel Details:\n{channels}"
455        )
